#                          INSTANCE MANAGEMENT - إدارة الـ Instances
################################################################################

--------------------------------------------------------------------------------
1. GET /whatsapp/instances
   الوصف: جلب جميع الـ instances الخاصة بالمستخدم
--------------------------------------------------------------------------------
   Request Headers:
      Authorization: Bearer {token}
   
   Request Body: لا يوجد
   
   Response (200):
   {
      "success": true,
      "data": [
         {
            "id": 1,
            "user_id": 1,
            "instance_name": "my-instance",
            "phone_number": "218921730546",
            "token": "xxx",
            "qr_code": "base64...",
            "status": "open|pending|close",
            "integration_type": "WHATSAPP-BAILEYS",
            "created_at": "2025-12-08T00:00:00.000Z",
            "updated_at": "2025-12-08T00:00:00.000Z"
         }
      ]
   }

--------------------------------------------------------------------------------
3. POST /whatsapp/instances
   الوصف: إنشاء instance جديد
--------------------------------------------------------------------------------
   Request Headers:
      Authorization: Bearer {token}
      Content-Type: application/json
   
   Request Body:
   {
      "name": "my-instance",              // required - اسم الـ instance (unique)
      "phone_number": "218921730546",     // required - رقم الهاتف
      "integration": "WHATSAPP-BAILEYS"   // optional - نوع التكامل
                                          // Values: WHATSAPP-BAILEYS, WHATSAPP-BUSINESS
   }
   
   Response (200):
   {
      "success": true,
      "data": {
         "id": 1,
         "instance_name": "my-instance",
         "phone_number": "218921730546",
         "token": "hash-token",
         "qr_code": "base64...",
         "status": "pending"
      }
   }
   
   Response (422):
   {
      "success": false,
      "message": "The name has already been taken."
   }

12. GET /whatsapp/instances/{instanceName}/messages
    الوصف: جلب الرسائل
--------------------------------------------------------------------------------
    Request Headers:
       Authorization: Bearer {token}
    
    Path Parameters:
       instanceName: string (required)
    
    Query Parameters:
       remote_jid: string (required) - معرف المحادثة (مثال: 218921730546@s.whatsapp.net)
       page: integer (optional) - رقم الصفحة
       limit: integer (optional) - عدد الرسائل (max: 100)
    
    Response (200):
    {
       "success": true,
       "data": [
          {
             "key": {
                "remoteJid": "218921730546@s.whatsapp.net",
                "fromMe": false,
                "id": "message-id"
             },
             "message": {
                "conversation": "text message"
             },
             "messageTimestamp": "1702000000"
          }
       ]
    }

25. POST /whatsapp/instances/{instanceId}/auto-reply/settings
    الوصف: تحديث إعدادات الرد التلقائي
--------------------------------------------------------------------------------
    Request Headers:
       Authorization: Bearer {token}
       Content-Type: application/json
    
    Path Parameters:
       instanceId: integer (required)
    
    Request Body (all optional):
    {
       "is_active": true,                    // boolean - تفعيل/تعطيل الرد التلقائي
       "stop_on_owner_message": false,       // boolean - إيقاف عند رسالة المالك
       "stop_on_keyword": false,             // boolean - إيقاف عند كلمة مفتاحية
       "stop_keywords": ["توقف", "stop"],   // array - كلمات الإيقاف
       "stop_duration": 60,                  // integer (min: 1) - مدة الإيقاف بالدقائق
       "custom_stop_duration": null,         // integer (min: 1) - مدة إيقاف مخصصة

       "show_typing": true,                  // boolean - إظهار "جاري الكتابة"
       "reply_once": false,                  // boolean - الرد مرة واحدة فقط
       "reply_delay": 2                      // integer (0-60) - تأخير الرد بالثواني
    }
    
    Response (200):
    {
       "success": true,
       "message": "تم تحديث الإعدادات بنجاح",
       "data": { ... }
    }
    
    Response (422):
    {
       "success": false,
       "message": "بيانات غير صالحة",
       "errors": { ... }
    }

29. POST /whatsapp/instances/{instanceId}/auto-reply/rules
    الوصف: إنشاء قاعدة جديدة
--------------------------------------------------------------------------------
    Request Headers:
       Authorization: Bearer {token}
       Content-Type: application/json
    
    Path Parameters:
       instanceId: integer (required)
    
    Request Body:
    {
       // إعدادات أساسية
       "name": "اسم القاعدة",                // required - string (max: 255)
       "is_active": true,                    // optional - boolean
       "priority": 0,                        // optional - integer (min: 0)
       
       // إعدادات المشغّل (Trigger)
       "trigger_type": "keyword",            // required - keyword|regex|all|contains
       "trigger_value": null,                // optional - string (للـ regex)
       "trigger_keywords": ["مرحبا", "هلا"], // optional - array
       "case_insensitive": true,             // optional - boolean (تجاهل حالة الأحرف)
       "exact_match": false,                 // optional - boolean (مطابقة تامة)
       
       // إعدادات الرد
       "response_type": "text",              // required - text|media|template|buttons
       "response_value": "الرد هنا",         // optional - string
       "random_response": false,             // optional - boolean (رد عشوائي)
       "alternative_responses": [            // optional - array (ردود بديلة للعشوائي)
          "رد 1",
          "رد 2"
       ],
       
       // إعدادات الوسائط (إذا response_type = media)
       "media_type": "image",                // optional - image|video|document|audio
       "media_url": "https://...",           // optional - string (max: 500)
       "media_caption": "وصف الوسائط",       // optional - string
       
       // إعدادات الأزرار (إذا response_type = buttons)
       "buttons_text": "نص الأزرار",         // optional - string
       "buttons": [                          // optional - array
          {"text": "نعم", "id": "yes"},
          {"text": "لا", "id": "no"}
       ],
       
       // إعدادات الجدولة
       "has_schedule": false,                // optional - boolean
       "schedule_start": "09:00:00",         // optional - time (H:i:s)
       "schedule_end": "17:00:00",           // optional - time (H:i:s)
       "schedule_days": [0, 1, 2, 3, 4]      // optional - array (0=الأحد ... 6=السبت)
